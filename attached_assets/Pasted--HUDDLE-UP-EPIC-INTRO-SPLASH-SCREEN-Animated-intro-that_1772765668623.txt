## 🎯 THE VISION:

**User logs in:**
1. Screen fades to black
2. ⚡ Huddle Up shield logo flies in with glow
3. 💫 "FIND YOUR CREW." fades in word-by-word
4. 🎮 "WATCH THE GAME." fades in word-by-word
5. ✨ Everything pulses/glows
6. 🚀 Explodes into confetti

```
// IntroSplash.jsx
import React, { useState, useEffect } from 'react';
import confetti from 'canvas-confetti';
import './IntroSplash.css';

function IntroSplash({ onComplete }) {
  const [stage, setStage] = useState(0);
  const [showSkip, setShowSkip] = useState(false);
  
  useEffect(() => {
    // Check if user wants to skip intros
    const skipIntros = localStorage.getItem('skipIntros');
    if (skipIntros === 'true') {
      onComplete();
      return;
    }
    
    // Show skip button after 1 second
    setTimeout(() => setShowSkip(true), 1000);
    
    // Animation timeline
    const timeline = [
      { stage: 1, delay: 500 },   // Logo appears
      { stage: 2, delay: 1200 },  // "FIND YOUR CREW."
      { stage: 3, delay: 1800 },  // "WATCH THE GAME."
      { stage: 4, delay: 2400 },  // Pulse effect
      { stage: 5, delay: 2800 }   // Confetti
    ];
    
    timeline.forEach(({ stage, delay }) => {
      setTimeout(() => setStage(stage), delay);
    });
    
    // Fire confetti at end
    setTimeout(() => {
      confetti({
        particleCount: 50,
        spread: 70,
        origin: { y: 0.6 },
        colors: ['#1E90FF', '#FFD700']
      });
    }, 2800);
    
    // Complete intro
    setTimeout(() => {
      onComplete();
    }, 3000);
  }, [onComplete]);
  
  const handleSkip = () => {
    onComplete();
  };
  
  const handleNeverShow = () => {
    localStorage.setItem('skipIntros', 'true');
    onComplete();
  };
  
  return (
    <div className="intro-splash" onClick={handleSkip}>
      <div className="intro-background"></div>
      
      {/* Logo */}
      <div className={`intro-logo ${stage >= 1 ? 'visible' : ''}`}>
        <img src="/huddle-up-logo.svg" alt="Huddle Up" />
      </div>
      
      {/* Slogan Line 1 */}
      <div className={`intro-text line-1 ${stage >= 2 ? 'visible' : ''}`}>
        <span className="word">FIND</span>
        <span className="word delay-1">YOUR</span>
        <span className="word delay-2">CREW.</span>
      </div>
      
      {/* Slogan Line 2 */}
      <div className={`intro-text line-2 ${stage >= 3 ? 'visible' : ''}`}>
        <span className="word">WATCH</span>
        <span className="word delay-1">THE</span>
        <span className="word delay-2">GAME.</span>
      </div>
      
      {/* Glow effect */}
      <div className={`intro-glow ${stage >= 4 ? 'visible' : ''}`}></div>
      
      {/* Skip button */}
      {showSkip && (
        <div className="skip-controls">
          <button className="skip-btn" onClick={handleSkip}>
            Skip →
          </button>
          <button className="never-show-btn" onClick={handleNeverShow}>
            Don't show again
          </button>
        </div>
      )}
    </div>
  );
}

export default IntroSplash;

// App.jsx or Login.jsx
import { useState, useEffect } from 'react';
import IntroSplash from './components/IntroSplash';

function App() {
  const [showIntro, setShowIntro] = useState(false);
  const [introComplete, setIntroComplete] = useState(false);
  
  useEffect(() => {
    // Show intro after successful login
    const justLoggedIn = sessionStorage.getItem('justLoggedIn');
    if (justLoggedIn === 'true') {
      setShowIntro(true);
      sessionStorage.removeItem('justLoggedIn');
    } else {
      setIntroComplete(true);
    }
  }, []);
  
  const handleIntroComplete = () => {
    setShowIntro(false);
    setIntroComplete(true);
  };
  
  return (
    <>
      {showIntro && <IntroSplash onComplete={handleIntroComplete} />}
      {introComplete && <YourMainApp />}
    </>
  );
}

// In your login handler:
const handleLoginSuccess = () => {
  sessionStorage.setItem('justLoggedIn', 'true');
  navigate('/');
};
```

### VERSION 1: Particles Background

```css
.intro-background::before {
  content: '';
  position: absolute;
  inset: 0;
  background-image: radial-gradient(circle, rgba(255,255,255,0.1) 1px, transparent 1px);
  background-size: 50px 50px;
  animation: particlesFloat 20s linear infinite;

```

### VERSION 2: Typing Effect
Make text type out letter-by-letter:
```javascript
const typeText = (text, element, speed = 100) => {
  let i = 0;
  const interval = setInterval(() => {
    element.textContent += text[i];
    i++;
    if (i === text.length) clearInterval(interval);
  }, speed);
};
```

### VERSION 3: 3D Logo Spin
Make logo spin in 3D:
```css
.intro-logo.visible {

  }

    transform: rotateY(0deg) scale(1);
    opacity: 1;
  }
}
```
