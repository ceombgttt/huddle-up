### STEP 2: CREATE SCORE MONITORING SERVICE

Create file: `server/services/scoreMonitor.js`

```javascript
const axios = require('axios');
const cron = require('node-cron');

// Store last known scores to detect changes
const gameScores = new Map();

// Your API key from https://the-odds-api.com (free tier)
const ODDS_API_KEY = process.env.ODDS_API_KEY || 'your_api_key_here';

// Monitor scores every 10 seconds
function startScoreMonitoring() {
  cron.schedule('*/10 * * * * *', async () => {
    await checkLiveScores();
  });
  
  console.log('✅ Score monitoring started - checking every 10 seconds');
}

async function checkLiveScores() {
  try {
    // Get live NBA games (add more sports later)
    const nbaGames = await getLiveNBAScores();
    
    for (const game of nbaGames) {
      await checkGameForScoreChanges(game);
    }
  } catch (error) {
    console.error('Error checking scores:', error.message);
  }
}

async function getLiveNBAScores() {
  const response = await axios.get(
    'https://api.the-odds-api.com/v4/sports/basketball_nba/scores/',
    {
      params: {
        apiKey: ODDS_API_KEY,
        daysFrom: 1
      }
    }
  );
  
  // Filter only active games
  return response.data.filter(game => !game.completed && game.scores);
}

async function checkGameForScoreChanges(game) {
  const gameId = game.id;
  
  const currentScore = {
    homeScore: game.scores?.find(s => s.name === game.home_team)?.score || 0,
    awayScore: game.scores?.find(s => s.name === game.away_team)?.score || 0,
    homeTeam: game.home_team,
    awayTeam: game.away_team
  };
  
  const lastScore = gameScores.get(gameId);
  
  // First time seeing this game
  if (!lastScore) {
    gameScores.set(gameId, currentScore);
    return;
  }
  
  // HOME TEAM SCORED!
  if (currentScore.homeScore > lastScore.homeScore) {
    const pointsScored = currentScore.homeScore - lastScore.homeScore;
    await notifyTeamScore({
      team: currentScore.homeTeam,
      points: pointsScored,
      gameId: gameId,
      sport: 'NBA'
    });
  }
  
  // AWAY TEAM SCORED!
  if (currentScore.awayScore > lastScore.awayScore) {
    const pointsScored = currentScore.awayScore - lastScore.awayScore;
    await notifyTeamScore({
      team: currentScore.awayTeam,
      points: pointsScored,
      gameId: gameId,
      sport: 'NBA'
    });
  }
  
  // Update stored score
  gameScores.set(gameId, currentScore);
}

async function notifyTeamScore({ team, points, gameId, sport }) {
  console.log(`🎉 ${team} SCORED ${points} points!`);
  
  // Get team colors
  const colors = getTeamColors(team);
  
  // Get users following this team
  const users = await getUsersFollowingTeam(team);
  
  // Send celebration event to all users
  for (const user of users) {
    // Emit event to connected user (Socket.IO or similar)
    global.io?.to(user.socketId)?.emit('team_scored', {
      team,
      points,
      colors,
      scoringType: getScoringType(points, sport)
    });
  }
}

function getTeamColors(teamName) {
  const teamColors = {
    // NBA Teams
    'Los Angeles Lakers': ['#552583', '#FDB927'],
    'Miami Heat': ['#98002E', '#F9A01B'],
    'Golden State Warriors': ['#1D428A', '#FFC72C'],
    'Boston Celtics': ['#007A33', '#BA9653'],
    'Chicago Bulls': ['#CE1141', '#000000'],
    'Dallas Mavericks': ['#00538C', '#002B5E'],
    'Phoenix Suns': ['#1D1160', '#E56020'],
    'Milwaukee Bucks': ['#00471B', '#EEE1C6'],
    
    // Add more teams...
  };
  
  return teamColors[teamName] || ['#1E90FF', '#FFD700']; // Default: app colors
}

function getScoringType(points, sport) {
  if (sport === 'NBA') {
    if (points === 1) return 'free_throw';
    if (points === 2) return 'basket';
    if (points === 3) return 'three_pointer';
  }
  if (sport === 'NFL') {
    if (points === 3) return 'field_goal';
    if (points === 6) return 'touchdown';
    if (points === 7) return 'touchdown_conversion';
  }
  return 'goal';
}

async function getUsersFollowingTeam(teamName) {
  // Query your database
  const users = await db.collection('users')
    .where('favoriteTeams', 'array-contains', teamName)
    .where('scoreNotificationsEnabled', '==', true)
    .get();
  
  return users.docs.map(doc => ({
    id: doc.id,
    ...doc.data()
  }));
}

module.exports = { startScoreMonitoring };
```

### STEP 4: CREATE CONFETTI CELEBRATION COMPONENT

Create file: `src/components/ScoreCelebration.jsx`

```jsx
import React, { useEffect } from 'react';
import confetti from 'canvas-confetti';
import './ScoreCelebration.css';

function ScoreCelebration({ socket, userId }) {
  useEffect(() => {
    if (!socket) return;
    
    // Listen for score events
    socket.on('team_scored', (data) => {
      handleTeamScored(data);
    });
    
    return () => {
      socket.off('team_scored');
    };
  }, [socket]);
  
  const handleTeamScored = ({ team, points, colors, scoringType }) => {
    // Fire confetti
    fireScoreCelebration(colors, scoringType, points);
    
    // Show popup
    showScorePopup(team, points);
    
    // Play sound
    playScoreSound(scoringType);
    
    // Vibrate phone
    if (navigator.vibrate) {
      navigator.vibrate([100, 50, 100, 50, 100]);
    }
  };
  
  const fireScoreCelebration = (colors, scoringType, points) => {
    const duration = scoringType === 'three_pointer' ? 2500 : 2000;
    const animationEnd = Date.now() + duration;
    
    const interval = setInterval(() => {
      const timeLeft = animationEnd - Date.now();
      
      if (timeLeft <= 0) {
        return clearInterval(interval);
      }
      
      // THREE POINTER - Triple burst!
      if (scoringType === 'three_pointer') {
        confetti({
          particleCount: 100,
          startVelocity: 50,
          spread: 100,
          origin: { x: 0.5, y: 0.3 },
          colors: colors,
          shapes: ['circle'],
          scalar: 2
        });
      } else {
        // Standard celebration
        confetti({
          particleCount: 50,
          startVelocity: 45,
          spread: 80,
          origin: { x: Math.random(), y: 0.6 },
          colors: colors,
          scalar: 1.5
        });
      }
    }, 250);
  };
  
  const showScorePopup = (team, points) => {
    const popup = document.createElement('div');
    popup.className = 'score-popup';
    popup.innerHTML = `
      <div class="score-popup-content">
        <div class="score-icon">🎉</div>
        <div class="score-team">${team}</div>
        <div class="score-points">+${points} POINTS!</div>
      </div>
    `;
    
    document.body.appendChild(popup);
    
    setTimeout(() => popup.classList.add('show'), 10);
    
    setTimeout(() => {
      popup.classList.remove('show');
      setTimeout(() => popup.remove(), 300);
    }, 3000);
  };
  
  const playScoreSound = (scoringType) => {
    const sounds = {
      'three_pointer': '/sounds/swish.mp3',
      'basket': '/sounds/basket.mp3',
      'free_throw': '/sounds/swish-small.mp3'
    };
    
    const audio = new Audio(sounds[scoringType] || sounds.basket);
    audio.volume = 0.5;
    audio.play().catch(err => console.log('Sound play failed:', err));
  };
  
  return null; // This component doesn't render anything
}

export default ScoreCelebration;
```
