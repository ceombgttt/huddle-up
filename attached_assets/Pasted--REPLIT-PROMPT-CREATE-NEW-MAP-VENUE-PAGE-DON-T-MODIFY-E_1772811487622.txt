import { GoogleMap, Marker, useLoadScript } from '@react-google-maps/api';

function VenueMap({ venues, center, onMarkerClick }) {
  const { isLoaded } = useLoadScript({
    googleMapsApiKey: process.env.REACT_APP_GOOGLE_MAPS_KEY
  });
  
  if (!isLoaded) return <div>Loading map...</div>;
  
  return (
    <GoogleMap
      mapContainerStyle={{ width: '100%', height: '250px' }}
      center={center}
      zoom={13}
      options={{
        styles: darkMapStyle, // Custom dark theme
        disableDefaultUI: false,
        zoomControl: true,
        streetViewControl: false
      }}
    >
      {venues.map(venue => (
        <Marker
          key={venue.id}
          position={{ lat: venue.latitude, lng: venue.longitude }}
          onClick={() => onMarkerClick(venue)}
          icon={{
            url: '/icons/venue-pin.svg',
            scaledSize: new window.google.maps.Size(40, 40)
          }}
        />
      ))}
    </GoogleMap>
  );
}

// Near Me search
const handleNearMe = async () => {
  try {
    const position = await getUserLocation();
    const venues = await fetchVenuesNear(
      position.lat, 
      position.lng, 
      10 // 10 mile radius
    );
    setVenues(venues);
    setMapCenter({ lat: position.lat, lng: position.lng });
  } catch (error) {
    alert('Please enable location access');
  }
};

// Get user location
const getUserLocation = () => {
  return new Promise((resolve, reject) => {
    navigator.geolocation.getCurrentPosition(
      (position) => resolve({
        lat: position.coords.latitude,
        lng: position.coords.longitude
      }),
      (error) => reject(error)
    );
  });
};

// City search
const handleCitySearch = async (cityName) => {
  try {
    const coords = await geocodeCity(cityName);
    const venues = await fetchVenuesNear(coords.lat, coords.lng, 25);
    setVenues(venues);
    setMapCenter(coords);
  } catch (error) {
    alert('City not found');
  }
};

// Geocode city name to coordinates
const geocodeCity = async (cityName) => {
  const response = await fetch(
    `https://maps.googleapis.com/maps/api/geocode/json?address=${cityName}&key=${GOOGLE_MAPS_KEY}`
  );
  const data = await response.json();
  return {
    lat: data.results[0].geometry.location.lat,
    lng: data.results[0].geometry.location.lng
  };
};

const getCategoryIcon = (category) => {
  const categories = {
    'Sports Bar': { icon: '🍺', color: '#FF6B35' },
    'Restaurant': { icon: '🍴', color: '#E63946' },
    'Pub': { icon: '🍻', color: '#10B981' },

    'Theater': { icon: '🎬', color: '#1E90FF' }
  };
  
  const { icon, color } = categories[category] || categories['Sports Bar'];
  
  return (
    <div className="category-icon-circle" style={{ background: color }}>
      {icon}
    </div>
  );
};
