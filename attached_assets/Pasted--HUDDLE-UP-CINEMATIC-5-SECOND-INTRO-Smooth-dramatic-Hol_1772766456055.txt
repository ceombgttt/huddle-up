CREATE CINEMATIC 5-SECOND INTRO:

TIMING (Smooth & Dramatic):
0.0s - 1.0s: Black screen → gradient background (slow fade)
1.0s - 2.0s: Logo rises from bottom with trail effect
2.0s - 3.0s: "FIND YOUR CREW." fades in (letter-by-letter, smooth)
3.0s - 4.0s: "WATCH THE GAME." fades in (letter-by-letter, smooth)
4.0s - 4.5s: Everything pulses with dramatic glow
4.5s - 5.0s: Smooth crossfade to home screen

VISUAL STYLE:
- Background: Animated gradient (deep blue → black → gold hints)
- Logo: Rises slowly, leaves light trail, glowing aura
- Text: Ultra-smooth fade-in, letter-by-letter reveal
- Particles: Floating golden particles in background
- Glow: Pulsing blue/gold radial glow behind everything
- No harsh cuts - everything flows smoothly

DRAMATIC EFFECTS:
- Logo has motion blur trail as it rises
- Text has soft glow around letters
- Background shifts colors subtly
- Particles float upward slowly
- Everything has easing (no linear animations)
- Sound: Deep bass hit when logo appears
- Sound: Soft chime when each word completes

SKIP CONTROLS:
- Subtle "Skip" text bottom-right (after 2 seconds)
- Tap anywhere to skip
- No "Don't show again" (we want users to see this!)

// CinematicIntro.jsx
import React, { useState, useEffect } from 'react';
import './CinematicIntro.css';

function CinematicIntro({ onComplete }) {
  const [stage, setStage] = useState(0);
  const [showSkip, setShowSkip] = useState(false);
  
  useEffect(() => {
    // Show skip after 2 seconds
    setTimeout(() => setShowSkip(true), 2000);
    
    // Smooth timeline
    const timeline = [
      { stage: 1, delay: 1000 },  // Logo rises
      { stage: 2, delay: 2000 },  // "FIND YOUR CREW."
      { stage: 3, delay: 3000 },  // "WATCH THE GAME."
      { stage: 4, delay: 4000 },  // Dramatic glow
      { stage: 5, delay: 4500 }   // Start fade out
    ];
    
    timeline.forEach(({ stage, delay }) => {
      setTimeout(() => setStage(stage), delay);
    });
    
    // Play sounds
    setTimeout(() => playSound('/sounds/logo-hit.mp3', 0.4), 1000);
    setTimeout(() => playSound('/sounds/text-chime.mp3', 0.3), 2000);
    setTimeout(() => playSound('/sounds/text-chime.mp3', 0.3), 3000);
    
    // Complete after 5 seconds
    setTimeout(() => {
      onComplete();
    }, 5000);
  }, [onComplete]);
  
  const playSound = (src, volume) => {
    const audio = new Audio(src);
    audio.volume = volume;
    audio.play().catch(() => {});
  };
  
  const handleSkip = (e) => {
    e.stopPropagation();
    onComplete();
  };
  
  return (
    <div className="cinematic-intro" onClick={handleSkip}>
      {/* Animated gradient background */}
      <div className="intro-bg">
        <div className="bg-gradient"></div>
        <div className="bg-particles"></div>
      </div>
      
      {/* Logo */}
      <div className={`intro-logo ${stage >= 1 ? 'visible' : ''}`}>
        <div className="logo-glow"></div>
        <div className="logo-trail"></div>
        <img src="/huddle-up-logo.svg" alt="Huddle Up" />
      </div>
      
      {/* Line 1: FIND YOUR CREW. */}
      <div className={`intro-line line-1 ${stage >= 2 ? 'visible' : ''}`}>
        <span className="letter" style={{transitionDelay: '0ms'}}>F</span>
        <span className="letter" style={{transitionDelay: '50ms'}}>I</span>
        <span className="letter" style={{transitionDelay: '100ms'}}>N</span>
        <span className="letter" style={{transitionDelay: '150ms'}}>D</span>
        <span className="space"> </span>
        <span className="letter" style={{transitionDelay: '200ms'}}>Y</span>
        <span className="letter" style={{transitionDelay: '250ms'}}>O</span>
        <span className="letter" style={{transitionDelay: '300ms'}}>U</span>
        <span className="letter" style={{transitionDelay: '350ms'}}>R</span>
        <span className="space"> </span>
        <span className="letter" style={{transitionDelay: '400ms'}}>C</span>
        <span className="letter" style={{transitionDelay: '450ms'}}>R</span>
        <span className="letter" style={{transitionDelay: '500ms'}}>E</span>
        <span className="letter" style={{transitionDelay: '550ms'}}>W</span>
        <span className="letter" style={{transitionDelay: '600ms'}}>.</span>
      </div>
      
      {/* Line 2: WATCH THE GAME. */}
      <div className={`intro-line line-2 ${stage >= 3 ? 'visible' : ''}`}>
        <span className="letter" style={{transitionDelay: '0ms'}}>W</span>
        <span className="letter" style={{transitionDelay: '50ms'}}>A</span>
        <span className="letter" style={{transitionDelay: '100ms'}}>T</span>
        <span className="letter" style={{transitionDelay: '150ms'}}>C</span>
        <span className="letter" style={{transitionDelay: '200ms'}}>H</span>
        <span className="space"> </span>
        <span className="letter" style={{transitionDelay: '250ms'}}>T</span>
        <span className="letter" style={{transitionDelay: '300ms'}}>H</span>
        <span className="letter" style={{transitionDelay: '350ms'}}>E</span>
        <span className="space"> </span>
        <span className="letter" style={{transitionDelay: '400ms'}}>G</span>
        <span className="letter" style={{transitionDelay: '450ms'}}>A</span>
        <span className="letter" style={{transitionDelay: '500ms'}}>M</span>
        <span className="letter" style={{transitionDelay: '550ms'}}>E</span>
        <span className="letter" style={{transitionDelay: '600ms'}}>.</span>
      </div>
      
      {/* Dramatic glow effect */}
      <div className={`dramatic-glow ${stage >= 4 ? 'active' : ''}`}></div>
      
      {/* Fade out overlay */}
      <div className={`fade-out ${stage >= 5 ? 'active' : ''}`}></div>
      
      {/* Skip button */}
      {showSkip && (
        <button className="skip-btn" onClick={handleSkip}>
          Skip
        </button>
      )}
    </div>
  );
}
