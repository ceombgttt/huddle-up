import React from 'react';
import './SportFilterCards.css';

const SPORTS = [
  { id: 'nfl', name: 'Football', image: '/images/sports/football.jpg', emoji: '🏈' },
  { id: 'nba', name: 'Basketball', image: '/images/sports/basketball.jpg', emoji: '🏀' },
  { id: 'mlb', name: 'Baseball', image: '/images/sports/baseball.jpg', emoji: '⚾' },
  { id: 'nhl', name: 'Hockey', image: '/images/sports/hockey.jpg', emoji: '🏒' },
  { id: 'soccer', name: 'Soccer', image: '/images/sports/soccer.jpg', emoji: '⚽' },
  { id: 'tennis', name: 'Tennis', image: '/images/sports/tennis.jpg', emoji: '🎾' },
  { id: 'golf', name: 'Golf', image: '/images/sports/golf.jpg', emoji: '⛳' },
  { id: 'mma', name: 'MMA', image: '/images/sports/mma.jpg', emoji: '🥊' },
  { id: 'nascar', name: 'NASCAR', image: '/images/sports/nascar.jpg', emoji: '🏎️' }
];

function SportFilterCards({ selectedSport, onSelectSport }) {
  return (
    <div className="sport-filters">
      <h2 className="filter-title">CHOOSE YOUR SPORT</h2>
      
      <div className="sport-cards-grid">
        {SPORTS.map(sport => (
          <div
            key={sport.id}
            className={`sport-card ${selectedSport === sport.id ? 'active' : ''}`}
            onClick={() => onSelectSport(sport.id)}
          >
            {/* Background Image */}
            <div 
              className="sport-card-bg"
              style={{ backgroundImage: `url(${sport.image})` }}
            />
            
            {/* Dark Overlay for readability */}
            <div className="sport-card-overlay" />
            
            {/* Content */}
            <div className="sport-card-content">
              <span className="sport-emoji">{sport.emoji}</span>
              <span className="sport-name">{sport.name}</span>
            </div>
            
            {/* Active indicator */}
            {selectedSport === sport.id && (
              <div className="sport-card-active-ring" />
            )}
          </div>
        ))}
      </div>
    </div>
  );
}

export default SportFilterCards;

STEP 4: GAME CARD WITH BACKGROUND

File: src/components/GameCard.jsx

import React from 'react';
import './GameCard.css';

function GameCard({ game, onClick }) {
  // Get sport background image
  const getSportBackground = (sport) => {
    const sportMap = {
      'NBA': 'basketball',
      'NFL': 'football',
      'MLB': 'baseball',
      'NHL': 'hockey',
      'MLS': 'soccer',
      'EPL': 'soccer',
      'Champions League': 'soccer',
      'Tennis': 'tennis',
      'Golf': 'golf',
      'MMA': 'mma',
      'NASCAR': 'nascar'
    };
    
    const sportKey = sportMap[game.sport] || 'basketball';
    return `/images/sports/${sportKey}.jpg`;
  };

  return (
    <div className="game-card" onClick={onClick}>
      {/* Background Image (subtle) */}
      <div 
        className="game-card-bg"
        style={{ backgroundImage: `url(${getSportBackground(game.sport)})` }}
      />
      
      {/* Overlay for readability */}
      <div className="game-card-overlay" />
      
      {/* Content */}
      <div className="game-card-content">
        {/* Header */}
        <div className="game-card-header">
          <span className="sport-badge">{game.sport}</span>
          {game.isLive && <span className="live-badge">🔴 LIVE</span>}
        </div>
        
        {/* Game Title */}
        <h3 className="game-title">
          {game.homeTeam} vs {game.awayTeam}
        </h3>
        
        {/* Venue */}
        <p className="venue-name">
          📍 {game.venueName}
        </p>
        
        {/* Time & Attendees */}
        <div className="game-card-footer">
          <span className="game-time">{game.time}</span>
          <span className="attendees">{game.attendees}👤 going</span>
        </div>
      </div>
    </div>
  );
}

export default GameCard;

STEP 7: OPACITY CONTROLS (Fine-tune readability)

If your backgrounds are too bright or too dark, adjust these values:

In GameCard.css:

/* Make background MORE visible (20-30%) */
.game-card-bg {
  opacity: 0.25; /* Change from 0.15 to 0.25 */
}

/* Make background LESS visible (5-10%) */
.game-card-bg {
  opacity: 0.08; /* Change from 0.15 to 0.08 */
}

/* Adjust gradient overlay darkness */
.game-card-overlay {
  background: linear-gradient(
    135deg,
    rgba(26, 29, 41, 0.9) 0%,  /* Darker = more opaque */
    rgba(15, 17, 21, 0.95) 100%
  );
}

STEP 9: FALLBACK FOR MISSING IMAGES

Add this to handle missing sport images:

In SportFilterCards.jsx, add error handling:

<div 
  className="sport-card-bg"
  style={{ 
    backgroundImage: `url(${sport.image})`,
    backgroundColor: '#1a1d29' // Fallback color
  }}
  onError={(e) => {
    // If image fails to load, use solid color
    e.target.style.backgroundImage = 'none';
    e.target.style.backgroundColor = '#2a2d39';
  }}
/>
