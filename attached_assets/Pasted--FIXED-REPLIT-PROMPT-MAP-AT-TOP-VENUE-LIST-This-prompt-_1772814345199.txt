MAP COMPONENT: src/components/VenueMap.jsx

import React from 'react';
import { GoogleMap, Marker, useLoadScript } from '@react-google-maps/api';

function VenueMap({ venues, center, onMarkerClick }) {
  const { isLoaded, loadError } = useLoadScript({
    googleMapsApiKey: process.env.REACT_APP_GOOGLE_MAPS_KEY || 'YOUR_API_KEY_HERE'
  });

  if (loadError) {
    return (
      <div style={{ 
        width: '100%', 
        height: '100%', 
        display: 'flex', 
        alignItems: 'center', 
        justifyContent: 'center',
        background: '#1a1d29',
        color: '#888'
      }}>
        Error loading map. Please check your API key.
      </div>
    );
  }

  if (!isLoaded) {
    return (
      <div style={{ 
        width: '100%', 
        height: '100%', 
        display: 'flex', 
        alignItems: 'center', 
        justifyContent: 'center',
        background: '#1a1d29',
        color: '#888'
      }}>
        Loading map...
      </div>
    );
  }

  const mapStyles = [
    {
      "elementType": "geometry",
      "stylers": [{ "color": "#1a1d29" }]
    },
    {
      "elementType": "labels.text.fill",
      "stylers": [{ "color": "#8a8a8a" }]
    },
    {
      "elementType": "labels.text.stroke",
      "stylers": [{ "color": "#1a1d29" }]
    },
    {
      "featureType": "water",
      "elementType": "geometry",
      "stylers": [{ "color": "#0f1115" }]
    }
  ];

  return (
    <GoogleMap
      mapContainerStyle={{ 
        width: '100%', 
        height: '100%' 
      }}
      center={center}
      zoom={13}
      options={{
        styles: mapStyles,
        disableDefaultUI: false,
        zoomControl: true,
        streetViewControl: false,
        mapTypeControl: false,
        fullscreenControl: false
      }}
    >
      {venues.map(venue => (
        <Marker
          key={venue.id}
          position={{ 
            lat: venue.latitude, 
            lng: venue.longitude 
          }}
          onClick={() => onMarkerClick(venue)}
          icon={{
            path: window.google.maps.SymbolPath.CIRCLE,
            fillColor: '#1E90FF',
            fillOpacity: 1,
            strokeColor: '#ffffff',
            strokeWeight: 2,
            scale: 10
          }}
        />
      ))}
    </GoogleMap>
  );
}

export default VenueMap;

VENUE CARD: src/components/VenueCard.jsx

import React from 'react';
import './VenueCard.css';

function VenueCard({ venue, onClick }) {
  const renderStars = (rating) => {
    return Array.from({ length: 5 }, (_, i) => (
      <span 
        key={i} 
        style={{ 
          color: i < rating ? '#FFD700' : '#444',
          fontSize: '14px'
        }}
      >
        ★
      </span>
    ));
  };

  const getCategoryIcon = (category) => {
    const icons = {
      'Sports Bar': { icon: '🍺', color: '#FF6B35' },
      'Pub': { icon: '🍻', color: '#10B981' },
      'Restaurant': { icon: '🍴', color: '#E63946' },
      'Lounge': { icon: '🎵', color: '#8B5CF6' }
    };
    return icons[category] || icons['Sports Bar'];
  };

  const { icon, color } = getCategoryIcon(venue.category);

  return (
    <div className="venue-card" onClick={onClick}>
      <div className="venue-image">
        <img src={venue.image} alt={venue.name} />
      </div>
      
      <div className="venue-info">
        <h3 className="venue-name">{venue.name}</h3>
        <p className="venue-category">{venue.category} • {venue.distance}</p>
        <div className="venue-rating">
          {renderStars(venue.rating)}
        </div>
        <p className="venue-stats">
          {venue.screens} Screens | {venue.attendees}👤00
        </p>
      </div>
      
      <div className="category-icon">
        <div 
          className="category-icon-circle"
          style={{ background: color }}
        >
          {icon}
        </div>
      </div>
    </div>
  );
}

export default VenueCard;
