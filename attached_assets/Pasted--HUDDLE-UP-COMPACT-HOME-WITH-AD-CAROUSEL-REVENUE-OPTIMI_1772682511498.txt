## AD CAROUSEL COMPONENT (SWIPEABLE)

```jsx
import React, { useState } from 'react';
import './AdCarousel.css';

function AdCarousel({ ads = [] }) {
  const [currentSlide, setCurrentSlide] = useState(0);
  
  // If no ads, show "Become a sponsor" placeholder
  const displayAds = ads.length > 0 ? ads : [
    {
      id: 'slot1',
      title: 'Sponsor Slot 1',
      subtitle: 'Reach thousands of sports fans',
      available: true,
      sport: 'NBA'
    },
    {
      id: 'slot2',
      title: 'Sponsor Slot 2',
      subtitle: 'Become a sponsor today',
      available: true,
      sport: 'NFL'
    },
    {
      id: 'slot3',
      title: 'Sponsor Slot 3',
      subtitle: '$2,500/month',
      available: true,
      sport: 'NHL'
    },
    {
      id: 'main',
      title: 'Main Homepage Spot',
      subtitle: '$20,000/month - Premium placement',
      available: true,
      featured: true
    }
  ];
  
  const handleSwipe = (direction) => {
    if (direction === 'left') {
      setCurrentSlide((prev) => (prev + 1) % displayAds.length);
    } else {
      setCurrentSlide((prev) => (prev - 1 + displayAds.length) % displayAds.length);
    }
  };
  
  return (
    <div className="ad-carousel-container">
      <div className="ad-carousel-hint">Swipe to see more ads</div>
      
      <div 
        className="ad-carousel"
        onTouchStart={(e) => {
          const touch = e.touches[0];
          this.startX = touch.clientX;
        }}
        onTouchEnd={(e) => {
          const touch = e.changedTouches[0];
          const diff = this.startX - touch.clientX;
          if (Math.abs(diff) > 50) {
            handleSwipe(diff > 0 ? 'left' : 'right');
          }
        }}
      >
        <div 
          className="ad-slides"
          style={{ transform: `translateX(-${currentSlide * 100}%)` }}
        >
          {displayAds.map((ad, index) => (
            <AdSlide 
              key={ad.id} 
              ad={ad} 
              active={index === currentSlide}
            />
          ))}
        </div>
      </div>
      
      <div className="ad-pagination">
        {displayAds.map((_, index) => (
          <span 
            key={index}
            className={`dot ${index === currentSlide ? 'active' : ''}`}
            onClick={() => setCurrentSlide(index)}
          />
        ))}
      </div>
    </div>
  );
}

function AdSlide({ ad, active }) {
  return (
    <div className={`ad-slide ${ad.featured ? 'featured' : ''} ${ad.available ? 'available' : ''}`}>
      {ad.available && (
        <div className="ad-badge">AVAILABLE</div>
      )}
      
      <div className="ad-icon">
        {ad.available ? '🏟️' : ad.brand_logo}
      </div>
      
      <h3 className="ad-title">{ad.title}</h3>
      <p className="ad-subtitle">{ad.subtitle}</p>
      
      {ad.available ? (
        <button className="ad-cta" onClick={() => window.location.href = '/advertise'}>
          Learn More →
        </button>
      ) : (
        <button className="ad-cta" onClick={() => window.open(ad.link)}>
          {ad.cta_text || 'Shop Now →'}
        </button>
      )}
    </div>
  );
}

export default AdCarousel;
```

## AD ROTATION LOGIC

**Show different ads based on:**

1. **Sport being viewed**
   - Viewing NBA content → Show NBA sponsor slot
   - Viewing NFL content → Show NFL sponsor slot

2. **User location**
   - Boca Raton users → Show local restaurant ads
   - Miami users → Show Miami-specific sponsors

3. **Time of day**
   - Morning: Show coffee/breakfast sponsors
   - Afternoon: Show lunch deals
   - Evening: Show bar/restaurant sponsors

4. **User behavior**
   - Frequent party creator → Show venue upgrade ads
   - Frequent joiner → Show fan subscription ads

```jsx
function getRelevantAd(context) {
  const { sport, location, timeOfDay, userType } = context;
  
  // Prioritize sport-specific ads
  if (sport === 'NBA') {
    return ads.filter(ad => ad.sport === 'NBA')[0];
  }
  
  // Fall back to location-based
  if (location) {
    return ads.filter(ad => ad.location === location)[0];
  }
  
  // Default to main homepage spot
  return ads.mainSpot;
}
```
