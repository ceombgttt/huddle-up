# VENUE QR CODE SIGNUP - INSTANT CONNECTION SYSTEM

**Venues create QR codes → Users scan → Instant "friends" → Direct party invites!** 🎯

---

## 🎯 WHAT THIS DOES:

1. **Venue generates unique QR code** for their location
2. **User scans QR code** on their phone
3. **User signs up** (or links if already registered)
4. **INSTANT CONNECTION** - User becomes venue's "contact"
5. **Venue can now:**
   - See user in their contacts list
   - Invite user to parties instantly
   - Send direct notifications
   - Build customer database

---

```sql
CREATE TABLE venue_contacts (
  id UUID PRIMARY KEY,
  venue_id UUID REFERENCES venues(id),
  user_id UUID REFERENCES users(id),
  connected_at TIMESTAMP DEFAULT NOW(),
  signup_source VARCHAR(50) DEFAULT 'qr_code',  -- 'qr_code', 'manual', 'check_in'
  last_visit TIMESTAMP,
  total_visits INTEGER DEFAULT 0,
  favorite BOOLEAN DEFAULT FALSE,
  notes TEXT,
  
  UNIQUE(venue_id, user_id)  -- Prevent duplicates
);

-- UPDATE USERS TABLE: Add referral code
ALTER TABLE users ADD COLUMN referred_by_venue_id UUID REFERENCES venues(id);
ALTER TABLE users ADD COLUMN signup_qr_code VARCHAR(255);  -- Which QR they scanned

-- UPDATE VENUES TABLE: Add QR code
ALTER TABLE venues ADD COLUMN signup_qr_code VARCHAR(255) UNIQUE;  -- Unique code per venue
ALTER TABLE venues ADD COLUMN qr_code_enabled BOOLEAN DEFAULT TRUE;
ALTER TABLE venues ADD COLUMN total_signups_via_qr INTEGER DEFAULT 0;

-- INDEX for fast lookups
CREATE INDEX idx_venue_contacts_venue ON venue_contacts(venue_id);
CREATE INDEX idx_venue_contacts_user ON venue_contacts(user_id);
CREATE INDEX idx_users_signup_qr ON users(signup_qr_code);
```

```jsx
import React, { useState, useEffect } from 'react';
import QRCode from 'qrcode.react';

function VenueQRCodeGenerator({ venueId }) {
  const [qrData, setQrData] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadQRCode();
  }, [venueId]);

  const loadQRCode = async () => {
    try {
      const response = await fetch(`/api/venues/${venueId}/qr-code`, {
        headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` }
      });
      
      if (response.status === 404) {
        // No QR code yet - generate one
        await generateQRCode();
        return;
      }
      
      const data = await response.json();
      setQrData(data);
    } catch (error) {
      console.error('Error loading QR code:', error);
    } finally {
      setLoading(false);
    }
  };

  const generateQRCode = async () => {
    try {
      const response = await fetch(`/api/venues/${venueId}/generate-qr-code`, {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` }
      });

      setQrData(data);

      console.error('Error generating QR code:', error);
```

```jsx
import React, { useState, useEffect } from 'react';

function QRSignup() {
  const [searchParams] = useSearchParams();
  const navigate = useNavigate();
  const qrCode = searchParams.get('venue');
  
  const [venueInfo, setVenueInfo] = useState(null);
  const [loading, setLoading] = useState(true);
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    password: '',
    confirmPassword: ''
  });

  useEffect(() => {
    if (qrCode) {
      verifyQRCode();
    }
  }, [qrCode]);

  const verifyQRCode = async () => {
    try {
      const response = await fetch(`/api/verify-qr/${qrCode}`);
      const data = await response.json();
      
      if (response.ok) {
        setVenueInfo(data.venue);
      } else {
        alert('Invalid QR code');
        navigate('/');
      }
    } catch (error) {
      console.error('Error verifying QR:', error);
      alert('Error verifying QR code');
    } finally {
      setLoading(false);
    }
  };

  const handleSignup = async (e) => {
    e.preventDefault();
    
    if (formData.password !== formData.confirmPassword) {
      alert('Passwords do not match');
      return;

    try {
      const response = await fetch('/api/signup-via-qr', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          qrCode,
          ...formData
        })

        // Save token
        localStorage.setItem('token', data.token);
        
        // Show success message
        alert(`Welcome! You're now connected with ${venueInfo.name}`);

        alert(data.error || 'Signup failed');

      console.error('Signup error:', error);
      alert('Error during signup');

  };

  if (loading) {
```

```jsx
    <div className="venue-contacts">
      <div className="contacts-header">
        <h2>Your Contacts</h2>
        <p>{contacts.length} connected customers</p>
      </div>

      <div className="contacts-search">
        <input
          type="text"
          placeholder="Search contacts..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
        />
      </div>

      <div className="contacts-list">
        {filteredContacts.map(contact => (
          <div key={contact.id} className="contact-card">
            <img
              src={contact.user.photo || '/default-avatar.png'}
              alt={contact.user.name}
              className="contact-photo"
            />
            
            <div className="contact-info">
              <h3>{contact.user.name}</h3>
              <p className="contact-email">{contact.user.email}</p>
              <p className="contact-meta">
                Connected {new Date(contact.connectedAt).toLocaleDateString()}
              </p>
              {contact.totalVisits > 0 && (
                <p className="contact-visits">{contact.totalVisits} visits</p>
              )}
            </div>

            <button
              className="btn-invite"
              onClick={() => {/* Show party selector */}}
            >
              Invite to Party
            </button>
          </div>
        ))}
      </div>

      {filteredContacts.length === 0 && (
        <div className="no-contacts">
          <p>No contacts yet. Share your QR code to get started!</p>
        </div>
      )}
    </div>
  );
}

export default VenueContacts;
```
