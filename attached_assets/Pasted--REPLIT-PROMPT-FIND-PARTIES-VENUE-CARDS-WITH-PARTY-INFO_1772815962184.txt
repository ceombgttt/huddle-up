import React, { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { GoogleMap, Marker, useLoadScript } from '@react-google-maps/api';
import './FindPartiesPage.css';

function FindPartiesPage() {
  const navigate = useNavigate();
  const [venuesWithParties, setVenuesWithParties] = useState([]);
  const [mapCenter, setMapCenter] = useState({ lat: 26.3683, lng: -80.1289 });
  const [loading, setLoading] = useState(true);

  const { isLoaded } = useLoadScript({
    googleMapsApiKey: process.env.REACT_APP_GOOGLE_MAPS_KEY || ''
  });

  useEffect(() => {
    loadNearbyVenuesWithParties();
  }, []);

  const loadNearbyVenuesWithParties = async () => {
    try {
      const position = await getUserLocation();
      const data = await fetchVenuesWithParties(position.lat, position.lng, 10);
      setVenuesWithParties(data);
      setMapCenter(position);
    } catch (error) {
      const data = await fetchVenuesWithParties(26.3683, -80.1289, 25);
      setVenuesWithParties(data);
    }
    setLoading(false);
  };

  const handleNearMe = async () => {
    try {
      const position = await getUserLocation();
      const data = await fetchVenuesWithParties(position.lat, position.lng, 10);
      setVenuesWithParties(data);
      setMapCenter(position);
    } catch (error) {
      alert('Please enable location access');
    }
  };

  const handleCitySearch = () => {
    const city = prompt('Enter city name:');
    if (city) searchByCity(city);
  };

  const searchByCity = async (cityName) => {
    alert(`Searching ${cityName}...`);
  };

  const getUserLocation = () => {
    return new Promise((resolve, reject) => {
      navigator.geolocation.getCurrentPosition(
        (pos) => resolve({ lat: pos.coords.latitude, lng: pos.coords.longitude }),
        reject
      );
    });
  };

  const fetchVenuesWithParties = async (lat, lng, radius) => {
    // TODO: Replace with actual API call
    // GET /api/venues/with-parties?lat={lat}&lng={lng}&radius={radius}
    
    // Mock data - venues with their parties:
    return [
      {
        // VENUE INFO
        venueId: 'venue_1',
        venueName: 'Jones Sports Bar',
        venueCategory: 'Sports Bar',
        venueImage: 'https://via.placeholder.com/80',
        venueRating: 5,
        venueScreens: 36,
        distance: '1.2mi',
        latitude: lat + 0.01,
        longitude: lng + 0.01,
        
        // PARTY INFO (happening at this venue)
        partyId: 'party_1',
        gameTitle: 'Heat vs Lakers',
        sportIcon: '🏀',
        gameTime: 'Thu, Mar 5 • 7:00 PM',
        attendees: 23,
        isLive: true
      },
      {
        venueId: 'venue_2',
        venueName: 'Blue Buzz Tavern',
        venueCategory: 'Pub',
        venueImage: 'https://via.placeholder.com/80',
        venueRating: 5,
        venueScreens: 38,
        distance: '2.1mi',
        latitude: lat - 0.01,
        longitude: lng - 0.01,
        
        partyId: 'party_2',
        gameTitle: 'Newcastle vs Barcelona',
        sportIcon: '⚽',
        gameTime: 'Sat, Mar 7 • 4:00 PM',
        attendees: 15,
        isLive: false
      },
      {
        venueId: 'venue_3',
        venueName: 'Central City Tap',
        venueCategory: 'Sports Bar',
        venueImage: 'https://via.placeholder.com/80',
        venueRating: 4,
        venueScreens: 30,
        distance: '3.2mi',
        latitude: lat + 0.02,
        longitude: lng - 0.02,
        
        partyId: 'party_3',
        gameTitle: 'Cowboys vs Eagles',
        sportIcon: '🏈',
        gameTime: 'Sun, Mar 8 • 1:00 PM',
        attendees: 31,
        isLive: false
      }
    ];
  };

  const getCategoryIcon = (category) => {
    const icons = {
      'Sports Bar': { icon: '🍺', color: '#FF6B35' },
      'Pub': { icon: '🍻', color: '#10B981' },
      'Restaurant': { icon: '🍴', color: '#E63946' }
    };
    return icons[category] || icons['Sports Bar'];
  };

  const renderStars = (rating) => {
    return Array.from({ length: 5 }, (_, i) => (
      <span key={i} style={{ color: i < rating ? '#FFD700' : '#444' }}>★</span>
    ));
  };

  return (
    <div className="find-parties-page">
      {/* HEADER */}
      <header className="page-header">
        <button className="back-btn" onClick={() => navigate(-1)}>
          ← Back
        </button>
        <div className="header-logo">
          <img src="/huddle-up-logo.svg" alt="Logo" />
          <span>HUDDLE UP</span>
        </div>
        <div className="spacer"></div>
      </header>

      {/* MAP - SHOWS WHERE PARTIES ARE */}
      <div className="map-container">
        {isLoaded ? (
          <GoogleMap
            mapContainerStyle={{ width: '100%', height: '100%' }}
            center={mapCenter}
            zoom={13}
            options={{
              styles: [
                { elementType: 'geometry', stylers: [{ color: '#1a1d29' }] },
                { elementType: 'labels.text.fill', stylers: [{ color: '#8a8a8a' }] }
              ],
              disableDefaultUI: false,
              zoomControl: true
            }}
          >
            {venuesWithParties.map(item => (
              <Marker
                key={item.partyId}
                position={{ lat: item.latitude, lng: item.longitude }}
                onClick={() => navigate(`/party/${item.partyId}`)}
                icon={{
                  path: window.google.maps.SymbolPath.CIRCLE,
                  fillColor: item.isLive ? '#ff4757' : '#1E90FF',
                  fillOpacity: 1,
                  strokeColor: '#fff',
                  strokeWeight: 2,
                  scale: 12
                }}
              />
            ))}
          </GoogleMap>
        ) : (
          <div className="map-loading">Loading map...</div>
        )}
      </div>

      {/* BANNER */}
      <div className="banner">
        <div className="banner-overlay"></div>
        <h2 className="banner-text">THE BEST PLACE TO WATCH THE GAME!</h2>
      </div>

      {/* SEARCH */}
      <div className="search-controls">
        <button className="near-me-btn" onClick={handleNearMe}>
          📍 Near Me
        </button>
        <button className="city-btn" onClick={handleCitySearch}>
          🏙️ Enter City
        </button>
      </div>

      {/* VENUE LIST WITH PARTIES */}
      <div className="venues-list">
        {loading ? (
          <div className="loading">Finding parties...</div>
        ) : venuesWithParties.length === 0 ? (
          <div className="no-parties">No parties found nearby</div>
        ) : (
          venuesWithParties.map(item => {
            const { icon, color } = getCategoryIcon(item.venueCategory);
            
            return (
              <div 
                key={item.partyId}
                className="venue-party-card"
                onClick={() => navigate(`/party/${item.partyId}`)}
              >
                {/* VENUE SECTION */}
                <div className="venue-section">
                  {/* Venue Image */}
                  <div className="venue-image">
                    <img src={item.venueImage} alt={item.venueName} />
                  </div>

                  {/* Venue Info */}
                  <div className="venue-info">
                    <h3 className="venue-name">{item.venueName}</h3>
                    <p className="venue-category">
                      {item.venueCategory} • {item.distance}
                    </p>
                    <div className="venue-rating">
                      {renderStars(item.venueRating)}
                    </div>
                    <p className="venue-screens">
                      {item.venueScreens} Screens
                    </p>
                  </div>

                  {/* Category Icon */}
                  <div className="category-icon">
                    <div className="icon-circle" style={{ background: color }}>
                      {icon}
                    </div>
                  </div>
                </div>

                {/* DIVIDER */}
                <div className="card-divider"></div>

                {/* PARTY SECTION */}
                <div className="party-section">
                  <div className="sport-icon">{item.sportIcon}</div>
                  <div className="party-details">
                    <h4 className="game-title">{item.gameTitle}</h4>
                    <p className="game-time">{item.gameTime}</p>
                    <p className="attendees">{item.attendees}👤 going</p>
                  </div>
                  {item.isLive && (
                    <div className="live-badge">🔴 LIVE</div>
                  )}
                </div>
              </div>
            );
          })
        )}
      </div>
    </div>
  );
}

export default FindPartiesPage;
